cmake_minimum_required(VERSION 3.16)
project(PaintOpenGL LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(PaintOpenGL WIN32 src/main.cpp)

if(WIN32)
    find_library(SDL2_LIBRARY NAMES SDL2 PATH_SUFFIXES lib)
    if(NOT SDL2_LIBRARY)
        message(FATAL_ERROR "SDL2 library not found. Install SDL2 development package (e.g. vcpkg install sdl2:x64-windows).")
    endif()
    target_link_libraries(PaintOpenGL PRIVATE ${SDL2_LIBRARY} opengl32)
elseif(UNIX)
    find_library(SDL2_LIBRARY NAMES SDL2 SDL2-2.0 SDL2-2.0.so.0 PATHS /lib/x86_64-linux-gnu /usr/lib/x86_64-linux-gnu NO_DEFAULT_PATH)
    if(NOT SDL2_LIBRARY AND EXISTS "/lib/x86_64-linux-gnu/libSDL2-2.0.so.0")
        set(SDL2_LIBRARY "/lib/x86_64-linux-gnu/libSDL2-2.0.so.0")
    endif()
    find_library(OPENGL_LIBRARY NAMES GL OpenGL libGL.so.1 PATHS /lib/x86_64-linux-gnu /usr/lib/x86_64-linux-gnu NO_DEFAULT_PATH)
    if(NOT OPENGL_LIBRARY AND EXISTS "/lib/x86_64-linux-gnu/libGL.so.1")
        set(OPENGL_LIBRARY "/lib/x86_64-linux-gnu/libGL.so.1")
    endif()
    if(NOT SDL2_LIBRARY OR NOT OPENGL_LIBRARY)
        message(FATAL_ERROR "SDL2/OpenGL runtime library not found")
    endif()
    target_link_libraries(PaintOpenGL PRIVATE ${SDL2_LIBRARY} ${OPENGL_LIBRARY} dl pthread)
endif()
